# Optional, platform-specific dependencies.
# Install with: pip install -r requirements-optional.txt
# Pip will only install lines whose environment markers match your system
# (e.g. on Windows, only lines without a marker or with sys_platform == 'win32' are considered).

# OQS (post-quantum) key encapsulation for utils/encryptor.py PersonalQuantumEncryptor.
# Install from GitHub; liboqs C library must be installed first (or is auto-installed at runtime).
# Full instructions: https://github.com/open-quantum-safe/liboqs-python/blob/main/README.md
git+https://github.com/open-quantum-safe/liboqs-python.git

# macOS (OS X): Foundation/Cocoa for optional keychain and native integration.
# Not required for basic keyring usage (utils/encryptor.py works without it).
pyobjc-framework-Cocoa; sys_platform == 'darwin'

# Linux: D-Bus for optional Secret Service keyring lock (utils/encryptor.py).
# Not required for basic keyring get/set.
dbus-python; sys_platform == 'linux'

# Image-to-prompt backend model downloads (fast tagger / BLIP captioner).
huggingface_hub

# Fast tagger (WD-style ONNX inference).
onnxruntime

# Captioner preferred path (Transformers BLIP). Falls back to legacy captioner if missing.
# Also serves the image-to-prompt VLM backend (LLaVA), which needs >= 4.36 for
# LlavaForConditionalGeneration; BLIP works on any version from here up, so the
# floor costs the captioner nothing.
transformers >= 4.36

# Legacy/local BLIP captioner backend (blip/* modules), and the VLM backend.
torch
torchvision

# VLM backend, both optional:
#   accelerate    — loads the model without briefly holding the weights twice
#                   in RAM. Skipped automatically when absent.
#   bitsandbytes  — 4-bit quantisation, roughly a third of the VRAM at some
#                   cost to quality. Only needed if config vlm_load_in_4bit is
#                   true, which it is not by default.
# accelerate
# bitsandbytes

# Model Context Protocol front end (extensions/mcp_server.py). Only needed to
# expose this app to an MCP client; the import is lazy, so without it the app
# starts normally and the MCP server declines to run.
mcp
